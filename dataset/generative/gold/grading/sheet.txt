## 17 [zm_health_facilities, geonames_cities | hard] rows=1 cols=['area_km2']
Q: What is the area in square kilometres of the convex hull of all health facilities after removing the convex hull of all district hospitals?

  SELECT ST_ConvexHull(ST_Union_Agg(geometry)) AS geom
  FROM zm_health_facilities
),
district_hull AS (
  SELECT ST_ConvexHull(ST_Union_Agg(geometry)) AS geom
  FROM zm_health_facilities
  WHERE NAME2_ = 'District Hospital'
),
diff AS (
  SELECT ST_Difference(all_hull.geom, district_hull.geom) AS geom
  FROM all_hull, district_hull
)

FROM diff;

  FROM uk_police_crimes
  GROUP BY street_name, category

GROUP BY st.street_name, st.total
ORDER BY st.total DESC, st.street_name ASC
LIMIT 10;

## 363 [nl_municipalities, eu_nuts_regions | hard] rows=1 cols=['water', 'gemeentenaam', 'rn', 'cum_area_ha', 'centroid_lon', 'centroid_lat']
Q: For Dutch municipality/water polygons with positive total area, rank by total area in hectares within each water class and compute cumulative area. Return the first rank in each water class where cumulative area reaches 100,000 hectares, with the centroid longitude and latitude.
WITH base AS (
  SELECT gemeentenaam, water, oppervlakte_totaal_in_ha,
         ST_X(ST_Centroid(geometry)) AS lon,
         ST_Y(ST_Centroid(geometry)) AS lat
  FROM nl_municipalities
  WHERE oppervlakte_totaal_in_ha > 0
    AND water IN ('B','JA','NEE')
), ranked AS (
  SELECT *,
         ROW_NUMBER() OVER (PARTITION BY water ORDER BY oppervlakte_totaal_in_ha DESC, gemeentenaam) AS rn,
         SUM(oppervlakte_totaal_in_ha) OVER (PARTITION BY water ORDER BY oppervlakte_totaal_in_ha DESC, gemeentenaam ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_area_ha
  FROM base
), first_cross AS (
  SELECT *,
         ROW_NUMBER() OVER (PARTITION BY water ORDER BY rn) AS cross_rn
  FROM ranked
  WHERE cum_area_ha >= 100000
)
SELECT water, gemeentenaam, rn, cum_area_ha,
       ROUND(lon,4) AS centroid_lon,
       ROUND(lat,4) AS centroid_lat
FROM first_cross
WHERE cross_rn = 1
ORDER BY water;
-> [['NEE', 'Het Hogeland', '2', '181094', '6.532', '53.3787']]
