  SELECT UNNEST([
    struct_pack(component := 'base_fare', amount := fare_amount),
    struct_pack(component := 'surcharges', amount := extra + mta_tax + improvement_surcharge),
    struct_pack(component := 'tolls', amount := tolls_amount),
