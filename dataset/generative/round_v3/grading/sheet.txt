Q: A marine planning zone is drawn as a 250 km-radius disc centred at (-59.5598, 13.1815) near Barbados. Split the zone into land and water, reporting each part's area in km² rounded to 1 decimal, treating every country polygon as land.
WITH disc AS (
  SELECT ST_Buffer(ST_Transform(ST_Point(-59.5598, 13.1815), 'EPSG:4326', 'EPSG:32621', always_xy := true), 250000) AS geom
),
disc4326 AS (
  SELECT ST_Transform(geom, 'EPSG:32621', 'EPSG:4326', always_xy := true) AS geom FROM disc
),
land AS (
  SELECT ST_Union_Agg(ST_Transform(c.geometry, 'EPSG:4326', 'EPSG:32621', always_xy := true)) AS geom
  FROM countries c, disc4326
  WHERE ST_Intersects(c.geometry, disc4326.geom)
)
SELECT
  ROUND(ST_Area(ST_Intersection(disc.geom, land.geom)) / 1000000.0, 1) AS land_km2,
  ROUND(ST_Area(ST_Difference(disc.geom, land.geom)) / 1000000.0, 1) AS water_km2
FROM disc, land;
-> [['2385.3', '192705.0']]

Q: For each of the lakes Lago Argentino, Lake Pukaki and Tehek Lake, which country's centroid is nearest, and what is that distance in kilometres?
WITH lake_pts AS (
  SELECT name, ST_Centroid(geometry) AS pt
  FROM lakes
  WHERE name IN ('Lago Argentino', 'Lake Pukaki', 'Tehek Lake')
),
country_pts AS (
  SELECT name, ST_Centroid(geometry) AS pt
  FROM countries
)
SELECT l.name AS lake,
       c.name AS nearest_country,
       round(ST_Distance_Sphere(c.pt, l.pt) / 1000.0, 1) AS distance_km
FROM lake_pts AS l, LATERAL (
  SELECT name, pt
  FROM country_pts
  ORDER BY ST_Distance_Sphere(pt, l.pt), name
  LIMIT 1
) AS c
ORDER BY lake;
-> [['Lago Argentino', 'Falkland Is.', '949.3'], ['Lake Pukaki', 'New Zealand', '260.3']]
